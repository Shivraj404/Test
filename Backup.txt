DROP TABLE IF EXISTS RESULT;
DROP TABLE IF EXISTS STUD_MARKS;

CREATE TABLE STUD_MARKS (
    RollNo INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    TotalMarks DECIMAL(5,2) CHECK (TotalMarks BETWEEN 0 AND 100)
);

CREATE TABLE RESULT (
    RollNo INT PRIMARY KEY,
    Name VARCHAR(50),
    Class VARCHAR(30)
);

(1, 'Amit', 95.00),
(2, 'Neha', 82.00),
(3, 'Rahul', 68.00),
(4, 'Priya', 45.00),
(5, 'Karan', 32.00);

DROP PROCEDURE IF EXISTS PROC_GRADE;

DELIMITER //
CREATE PROCEDURE PROC_GRADE()
BEGIN
    DELETE FROM RESULT;
    INSERT INTO RESULT (RollNo, Name, Class)
    SELECT 
        RollNo, 
        Name,
        CASE
            WHEN TotalMarks >= 90 THEN 'Distinction'
            WHEN TotalMarks >= 75 THEN 'First Class'
            WHEN TotalMarks >= 60 THEN 'Second Class'
            WHEN TotalMarks >= 40 THEN 'Pass'
            ELSE 'Fail'
        END AS Class
    FROM STUD_MARKS;
END //
DELIMITER ;

-- Invoke procedure and display result
CALL PROC_GRADE();
SELECT * FROM RESULT ORDER BY RollNo;

    INSERT INTO LIBRARY_AUDIT

BEFORE DELETE ON LIBRARY
FOR EACH ROW
BEGIN
    INSERT INTO LIBRARY_AUDIT
    (BookID, BookName, Author, Price, OperationType, OperationDate)
    VALUES
    (OLD.BookID, OLD.BookName, OLD.Author, OLD.Price, 'DELETE', NOW());
END //

-- (1) INNER JOIN: matching employee and department rows

FROM EMPLOYEE e 
INNER JOIN DEPARTMENT d ON e.DeptID = d.DeptID;

-- (2) LEFT JOIN: all employees, including those without a department
SELECT e.EmpName, d.DeptName
FROM EMPLOYEE e 
LEFT JOIN DEPARTMENT d ON e.DeptID = d.DeptID;

-- (3) RIGHT JOIN: all departments, including departments with no employees
SELECT e.EmpName, d.DeptName
FROM EMPLOYEE e 
RIGHT JOIN DEPARTMENT d ON e.DeptID = d.DeptID;

-- (4) FULL OUTER JOIN equivalent in MySQL
SELECT e.EmpID, e.EmpName, d.DeptID, d.DeptName
FROM EMPLOYEE e LEFT JOIN DEPARTMENT d ON e.DeptID = d.DeptID
UNION ALL
SELECT e.EmpID, e.EmpName, d.DeptID, d.DeptName
FROM EMPLOYEE e RIGHT JOIN DEPARTMENT d ON e.DeptID = d.DeptID
WHERE e.EmpID IS NULL;


/* ============================================================================
   CHIT 4B - Merge New and Old Roll Call Records (MySQL Cursor Procedure)
   ============================================================================ */
DROP TABLE IF EXISTS O_ROLLCALL;
DROP TABLE IF EXISTS N_ROLLCALL;

CREATE TABLE O_ROLLCALL (
    RollNo INT PRIMARY KEY,
    StudentName VARCHAR(60),
    AttendanceStatus VARCHAR(20)
);

CREATE TABLE N_ROLLCALL (
    RollNo INT,
    StudentName VARCHAR(60),
    AttendanceStatus VARCHAR(20)
);

INSERT INTO O_ROLLCALL VALUES (1, 'Amit', 'Present');
INSERT INTO N_ROLLCALL VALUES 
(1, 'Amit', 'Present'), 
(2, 'Neha', 'Absent'), 
(3, 'Rahul', 'Present');

DROP PROCEDURE IF EXISTS MERGE_ROLLCALL;

DELIMITER //
CREATE PROCEDURE MERGE_ROLLCALL()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_roll INT;
    DECLARE v_name VARCHAR(60);
    DECLARE v_status VARCHAR(20);
    DECLARE v_exists INT;
    DECLARE v_inserted INT DEFAULT 0;

    DECLARE c_new CURSOR FOR 
        SELECT RollNo, StudentName, AttendanceStatus FROM N_ROLLCALL;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    OPEN c_new;
    read_loop: LOOP
        FETCH c_new INTO v_roll, v_name, v_status;
        IF done THEN
            LEAVE read_loop;
        END IF;

        SELECT COUNT(*) INTO v_exists 
        FROM O_ROLLCALL WHERE RollNo = v_roll;

        IF v_exists = 0 THEN
            INSERT INTO O_ROLLCALL (RollNo, StudentName, AttendanceStatus)
            VALUES (v_roll, v_name, v_status);
            SET v_inserted = v_inserted + 1;
        END IF;
    END LOOP;
    CLOSE c_new;

    SELECT CONCAT('Inserted records: ', v_inserted) AS StatusMessage;
END //
DELIMITER ;

CALL MERGE_ROLLCALL();
SELECT * FROM O_ROLLCALL ORDER BY RollNo;

    Credits INT CHECK (Credits > 0)

CREATE TABLE REGISTRATION (
    RollNo INT,
    CourseID INT,
    PRIMARY KEY (RollNo, CourseID),
    FOREIGN KEY (RollNo) REFERENCES STUDENT(RollNo),
    FOREIGN KEY (CourseID) REFERENCES COURSE(CourseID)
);

-- (1) Student-course details
SELECT s.RollNo, s.Name, c.CourseID, c.CourseName, c.Credits
FROM STUDENT s
JOIN REGISTRATION r ON r.RollNo = s.RollNo
JOIN COURSE c ON c.CourseID = r.CourseID;

-- (2) Students registered for a specified course (DBMS - 101)
SELECT s.RollNo, s.Name
FROM STUDENT s 
JOIN REGISTRATION r ON r.RollNo = s.RollNo
WHERE r.CourseID = 101;

-- (3) Students not registered for any course
SELECT s.* FROM STUDENT s
WHERE NOT EXISTS (
    SELECT 1 FROM REGISTRATION r WHERE r.RollNo = s.RollNo
);

-- (4) Number of students registered for each course
SELECT 
    c.CourseID, 
    c.CourseName, 
    COUNT(r.RollNo) AS StudentCount
FROM COURSE c 
LEFT JOIN REGISTRATION r ON r.CourseID = c.CourseID
GROUP BY c.CourseID, c.CourseName;

DROP PROCEDURE IF EXISTS PROC_EXPLICIT_CURSOR;

DELIMITER //
CREATE PROCEDURE PROC_EXPLICIT_CURSOR()
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_id INT;
    DECLARE v_name VARCHAR(50);
    DECLARE v_salary DECIMAL(10,2);
    DECLARE v_count INT DEFAULT 0;

    -- Explicit cursor for employees earning > 40,000
    DECLARE c_emp CURSOR FOR 
        SELECT EmpID, EmpName, Salary FROM EMPLOYEE WHERE Salary > 40000;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    DROP TEMPORARY TABLE IF EXISTS OUTPUT_TABLE;
    CREATE TEMPORARY TABLE OUTPUT_TABLE (
        EmpID INT,
        EmpName VARCHAR(50),
        Salary DECIMAL(10,2)
    );

    OPEN c_emp;
    read_loop: LOOP
        FETCH c_emp INTO v_id, v_name, v_salary;
        IF done THEN
            LEAVE read_loop;
        END IF;
        INSERT INTO OUTPUT_TABLE VALUES (v_id, v_name, v_salary);
        SET v_count = v_count + 1;
    END LOOP;
    CLOSE c_emp;

    -- Display retrieved records and total count
    SELECT * FROM OUTPUT_TABLE;
    SELECT CONCAT('Records processed: ', v_count) AS TotalCount;
END //
DELIMITER ;

CALL PROC_EXPLICIT_CURSOR();

DROP PROCEDURE IF EXISTS GET_DEPT_EMPLOYEES;

DELIMITER //
CREATE PROCEDURE GET_DEPT_EMPLOYEES(IN p_dept_id INT)
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_id INT;
    DECLARE v_name VARCHAR(50);
    DECLARE v_sal DECIMAL(10,2);

    -- Cursor accepting department ID
    DECLARE c_emp CURSOR FOR 
        SELECT EmpID, EmpName, Salary FROM EMPLOYEE WHERE DeptID = p_dept_id;
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    DROP TEMPORARY TABLE IF EXISTS DEPT_OUTPUT;
    CREATE TEMPORARY TABLE DEPT_OUTPUT (
        DeptID INT,
        EmpID INT,
        EmpName VARCHAR(50),
        Salary DECIMAL(10,2)
    );

    OPEN c_emp;
    read_loop: LOOP
        FETCH c_emp INTO v_id, v_name, v_sal;
        IF done THEN
            LEAVE read_loop;
        END IF;
        INSERT INTO DEPT_OUTPUT VALUES (p_dept_id, v_id, v_name, v_sal);
    END LOOP;
    CLOSE c_emp;

    SELECT * FROM DEPT_OUTPUT;
END //
DELIMITER ;

-- Execute the cursor for at least two department IDs (10 and 20)

/* ============================================================================
   CHIT 10B - Function to Calculate Tax (MySQL Stored Function)
   ============================================================================ */
/*
   Assumed Progressive Tax Slabs:
   - Up to Rs. 2,50,000         : 0%
   - Rs. 2,50,001 to 5,00,000   : 5%
   - Rs. 5,00,001 to 10,00,000  : 10%
   - Rs. 10,00,001 to 15,00,000 : 20%
   - Above Rs. 15,00,000        : 30%
*/
DROP FUNCTION IF EXISTS CALCULATE_TAX;

DELIMITER //
CREATE FUNCTION CALCULATE_TAX(p_income DECIMAL(14,2))
RETURNS DECIMAL(14,2)
DETERMINISTIC
BEGIN
    DECLARE v_tax DECIMAL(14,2) DEFAULT 0;

    IF p_income <= 250000 THEN
        SET v_tax = 0;
    ELSEIF p_income <= 500000 THEN
        SET v_tax = (p_income - 250000) * 0.05;
    ELSEIF p_income <= 1000000 THEN
        SET v_tax = 12500 + (p_income - 500000) * 0.10;
    ELSEIF p_income <= 1500000 THEN
        SET v_tax = 62500 + (p_income - 1000000) * 0.20;
    ELSE
        SET v_tax = 162500 + (p_income - 1500000) * 0.30;
    END IF;

    RETURN v_tax;
END //
DELIMITER ;

-- Invoke function for multiple income values
SELECT 200000 AS Income, CALCULATE_TAX(200000) AS Tax
UNION ALL
SELECT 450000, CALCULATE_TAX(450000)
UNION ALL
SELECT 800000, CALCULATE_TAX(800000)
UNION ALL
SELECT 1800000, CALCULATE_TAX(1800000);

)

/* ============================================================================
   CHIT 16A - University Examination Database (MySQL)
   ============================================================================ */
DROP TABLE IF EXISTS RESULT;
DROP TABLE IF EXISTS STUDENT;

CREATE TABLE STUDENT (
    RollNo INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Department VARCHAR(30)
);

CREATE TABLE RESULT (
    RollNo INT,
    Subject VARCHAR(40),
    Marks DECIMAL(5,2) CHECK (Marks BETWEEN 0 AND 100),
    PRIMARY KEY (RollNo, Subject),
    FOREIGN KEY (RollNo) REFERENCES STUDENT(RollNo)
);

INSERT INTO STUDENT VALUES (1, 'Amit', 'IT'), (2, 'Neha', 'CSE'), (3, 'Rahul', 'IT');
INSERT INTO RESULT VALUES
(1, 'DBMS', 75.00), (1, 'Java', 65.00),
(2, 'DBMS', 88.00), (2, 'Java', 92.00),
(3, 'DBMS', 35.00), (3, 'Java', 55.00);

-- (1) Student-wise total marks and average marks
SELECT 
    s.RollNo, 
    s.Name, 
    SUM(r.Marks) AS TotalMarks,
    AVG(r.Marks) AS AverageMarks
FROM STUDENT s 
JOIN RESULT r ON r.RollNo = s.RollNo
GROUP BY s.RollNo, s.Name;

-- (2) Subject-wise highest marks
SELECT 
    Subject, 
    MAX(Marks) AS HighestMarks
FROM RESULT 
GROUP BY Subject;

-- (3) Students who failed in at least one subject (passing threshold = 40)
SELECT DISTINCT s.RollNo, s.Name
FROM STUDENT s 
JOIN RESULT r ON r.RollNo = s.RollNo
WHERE r.Marks < 40;

-- (4) Students whose average marks exceed the overall average
SELECT 
    s.RollNo, 
    s.Name, 
    AVG(r.Marks) AS StudentAverage
FROM STUDENT s 
JOIN RESULT r ON r.RollNo = s.RollNo
GROUP BY s.RollNo, s.Name
HAVING AVG(r.Marks) > (SELECT AVG(Marks) FROM RESULT);

/* ============================================================================
   CHIT 17A - SQL Transactions (MySQL / InnoDB)
   ============================================================================ */
DROP TABLE IF EXISTS ACCOUNT;
CREATE TABLE ACCOUNT (
    AccountNo INT PRIMARY KEY,
    HolderName VARCHAR(50),
    Balance DECIMAL(12,2) CHECK (Balance >= 0)
) ENGINE=InnoDB;

INSERT INTO ACCOUNT VALUES (101, 'Amit', 5000.00), (102, 'Neha', 3000.00);

-- Part 1: Demonstrate SAVEPOINT & ROLLBACK
START TRANSACTION;
UPDATE ACCOUNT SET Balance = Balance - 1000.00 WHERE AccountNo = 101;
UPDATE ACCOUNT SET Balance = Balance + 1000.00 WHERE AccountNo = 102;

SAVEPOINT after_transfer;
UPDATE ACCOUNT SET Balance = Balance - 200.00 WHERE AccountNo = 101;
SELECT * FROM ACCOUNT ORDER BY AccountNo;

-- Rollback only to savepoint
ROLLBACK TO SAVEPOINT after_transfer;
SELECT * FROM ACCOUNT ORDER BY AccountNo;

-- Full rollback restores original balance
ROLLBACK;
SELECT * FROM ACCOUNT ORDER BY AccountNo;

-- Part 2: Final Committed Transfer
START TRANSACTION;
UPDATE ACCOUNT SET Balance = Balance - 1000.00 WHERE AccountNo = 101;
UPDATE ACCOUNT SET Balance = Balance + 1000.00 WHERE AccountNo = 102;
COMMIT;

SELECT * FROM ACCOUNT ORDER BY AccountNo;

   CHIT 18A - Inventory Management Using Nested Queries (MySQL)
   ============================================================================ */
DROP TABLE IF EXISTS INVENTORY;
CREATE TABLE INVENTORY (
    ItemID INT PRIMARY KEY,
    ItemName VARCHAR(60) NOT NULL,
    Category VARCHAR(40) NOT NULL,
    Price DECIMAL(10,2) CHECK (Price >= 0),
    Stock INT CHECK (Stock >= 0)
);

INSERT INTO INVENTORY VALUES
(1, 'Mouse', 'Accessories', 650.00, 30),
(2, 'Keyboard', 'Accessories', 1200.00, 15),
(3, 'Monitor', 'Electronics', 8500.00, 8),
(4, 'SSD', 'Storage', 3200.00, 20),
(5, 'USB Cable', 'Accessories', 250.00, 50),
(6, 'Laptop', 'Electronics', 55000.00, 5);

-- (1) Products priced above overall average
SELECT * FROM INVENTORY
WHERE Price > (SELECT AVG(Price) FROM INVENTORY);

-- (2) Most expensive product in each category (correlated subquery)
SELECT i.* FROM INVENTORY i
WHERE i.Price = (
    SELECT MAX(i2.Price) FROM INVENTORY i2
    WHERE i2.Category = i.Category
);

-- (3) Products with stock below overall average stock
SELECT * FROM INVENTORY
WHERE Stock < (SELECT AVG(Stock) FROM INVENTORY);

-- (4) Categories containing products costing more than Rs. 1,000
SELECT DISTINCT Category FROM INVENTORY
WHERE ItemID IN (
    SELECT ItemID FROM INVENTORY WHERE Price > 1000
);

/* ============================================================================
   CHIT 20A - Attendance Management Database (MySQL)
   ============================================================================ */
DROP TABLE IF EXISTS ATTENDANCE;
DROP TABLE IF EXISTS STUDENT;

CREATE TABLE STUDENT (
    RollNo INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Department VARCHAR(30)
);

CREATE TABLE ATTENDANCE (
    RollNo INT,
    Subject VARCHAR(40),
    TotalClasses INT CHECK (TotalClasses > 0),
    AttendedClasses INT CHECK (AttendedClasses >= 0),
    PRIMARY KEY (RollNo, Subject),
    FOREIGN KEY (RollNo) REFERENCES STUDENT(RollNo),
    CHECK (AttendedClasses <= TotalClasses)
);

INSERT INTO STUDENT VALUES (1, 'Amit', 'IT'), (2, 'Neha', 'CSE'), (3, 'Rahul', 'IT');
INSERT INTO ATTENDANCE VALUES
(1, 'DBMS', 40, 34), (1, 'Java', 40, 31),
(2, 'DBMS', 40, 28), (2, 'Java', 40, 32),
(3, 'DBMS', 40, 25), (3, 'Java', 40, 36);

-- (1) Calculate attendance percentage for each student and subject
SELECT 
    s.RollNo, 
    s.Name, 
    a.Subject,
    ROUND(100.0 * a.AttendedClasses / a.TotalClasses, 2) AS AttendancePercentage
FROM STUDENT s 
JOIN ATTENDANCE a ON a.RollNo = s.RollNo
ORDER BY s.RollNo, a.Subject;

-- (2) Identify students with attendance below 75%
SELECT 
    s.RollNo, 
    s.Name, 
    a.Subject,
    ROUND(100.0 * a.AttendedClasses / a.TotalClasses, 2) AS AttendancePercentage
FROM STUDENT s 
JOIN ATTENDANCE a ON a.RollNo = s.RollNo
WHERE (1.0 * a.AttendedClasses / a.TotalClasses) < 0.75;

-- (3) Display subject-wise average attendance percentage
SELECT 
    Subject,
    ROUND(AVG(100.0 * AttendedClasses / TotalClasses), 2) AS AverageAttendancePercentage
FROM ATTENDANCE 
GROUP BY Subject;

-- (4) Find students meeting the attendance threshold (>= 75%) in every subject
SELECT 
    s.RollNo, 
    s.Name
FROM STUDENT s 
JOIN ATTENDANCE a ON a.RollNo = s.RollNo
GROUP BY s.RollNo, s.Name
HAVING MIN(1.0 * a.AttendedClasses / a.TotalClasses) >= 0.75;


/* ============================================================================
   CHIT 20B - Sales Analysis Using Aggregation (MySQL & MongoDB)
   ============================================================================ */
-- [MySQL Implementation]
DROP TABLE IF EXISTS SALES;
CREATE TABLE SALES (
    SaleID INT AUTO_INCREMENT PRIMARY KEY,
    Product VARCHAR(60),
    Category VARCHAR(40),
    Quantity INT CHECK (Quantity >= 0),
    UnitPrice DECIMAL(10,2) CHECK (UnitPrice >= 0),
    SaleDate DATE
);

INSERT INTO SALES (Product, Category, Quantity, UnitPrice, SaleDate) VALUES
('Mouse', 'Accessories', 20, 650.00, '2026-10-01'),
('Keyboard', 'Accessories', 10, 1200.00, '2026-10-02'),
('Monitor', 'Electronics', 3, 8500.00, '2026-10-15'),
('SSD', 'Storage', 8, 3200.00, '2026-10-20'),
('Laptop', 'Electronics', 2, 55000.00, '2026-10-24');

-- Calculate total quantity and revenue per category, ranked by revenue descending
SELECT 
    Category,
    SUM(Quantity) AS TotalQuantity,
    SUM(Quantity * UnitPrice) AS TotalRevenue
FROM SALES
GROUP BY Category
ORDER BY TotalRevenue DESC;
